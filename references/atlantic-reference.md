document.addEventListener("DOMContentLoaded", function () {

  let currentIndex = 0;
  let currentGroup = [];

  // Open modal on click
  document.querySelectorAll(".gallery-image-container").forEach((img) => {
    img.addEventListener("click", () => {
      const groupName = img.getAttribute("data-group");

      // collect only images from the same block
      currentGroup = [...document.querySelectorAll(`.gallery-image-container[data-group="${groupName}"]`)];
      currentIndex = currentGroup.indexOf(img);

      showImage(currentIndex);
      modal.style.display = "flex";
    });
  });

  // Show image with lazy loading & responsive sizes
  function showImage(index) {
    const isMobile = window.innerWidth <= 768;
    const src = isMobile
      ? currentGroup[index].getAttribute("data-src-mobile") || currentGroup[index].getAttribute("data-src")
      : currentGroup[index].getAttribute("data-src-desktop") || currentGroup[index].getAttribute("data-src");

    modalImg.style.opacity = "0";  
    if (loader) loader.style.display = "block"; 

    const img = new Image();
    img.src = src;
    img.onload = () => {
      modalImg.src = src;
      modalImg.alt = "Full view image " + (index + 1);
      if (loader) loader.style.display = "none";
      modalImg.style.opacity = "1"; 
    };
  }

  // Navigation
  nextBtn?.addEventListener("click", () => {
    currentIndex = (currentIndex + 1) % currentGroup.length;
    showImage(currentIndex);
  });

  prevBtn?.addEventListener("click", () => {
    currentIndex = (currentIndex - 1 + currentGroup.length) % currentGroup.length;
    showImage(currentIndex);
  });

  // Close modal
  closeBtn?.addEventListener("click", () => modal.style.display = "none");
  modal.addEventListener("click", (e) => {
    if (e.target === modal) modal.style.display = "none";
  });

  // Keyboard support
  document.addEventListener("keydown", (e) => {
    if (modal.style.display === "flex") {
      if (e.key === "ArrowRight") {
        currentIndex = (currentIndex + 1) % currentGroup.length;
        showImage(currentIndex);
      } else if (e.key === "ArrowLeft") {
        currentIndex = (currentIndex - 1 + currentGroup.length) % currentGroup.length;
        showImage(currentIndex);
      } else if (e.key === "Escape") {
        modal.style.display = "none";
      }
    }
  });
});

<script>
  document.addEventListener('DOMContentLoaded', function () {
    // 1. Select all variant radio buttons
    const variantRadioButtons = document.querySelectorAll('input[name="Book+cover+type"]');

    // 2. References to DOM elements inside More Information tab
    const isbnElement = document.getElementById('variant-isbn13');
    const weightElement = document.getElementById('variant-weight');
    const priceElement = document.getElementById('variant-original-price');
    const editionElement = document.getElementById('variant-edition');
    const bindingElement = document.getElementById('variant-binding');

    // Helper: Standardize empty or "N/A" values to "N/A"
    function handleEmptyValue(value) {
      if (!value || value.trim() === '' || value.trim().toUpperCase() === 'N/A') {
        return 'N/A';
      }
      return value;
    }

    // 3. Parse the stored metafield JSON
    const metafieldDataEl = document.getElementById('variant_metafield_data');
    let metafieldData = {};
    if (metafieldDataEl && metafieldDataEl.textContent) {
      try {
        metafieldData = JSON.parse(metafieldDataEl.textContent);
      } catch (e) {
        console.error('Error parsing metafield JSON:', e);
      }
    }

    // --- Reusable function to update the More Information tab ---
    function updateVariantData(radioValue) {
      // 1) Get the product handle from URL (last part of /products/xxxx)
      const productHandle = window.location.pathname.split('/').pop();
      console.log('Handle', productHandle);

      // 2) Fetch the product data in JSON
      fetch(`/products/${productHandle}.js`)
        .then((response) => response.json())
        .then((data) => {
          // 3) Find the matching variant by comparing option1
          const selectedVariant = data.variants.find((v) => v.option1 === radioValue);
          if (!selectedVariant) return;

          // 4) Retrieve stored metafield data for this variant ID
          const variantMeta = metafieldData[selectedVariant.id];
          if (!variantMeta) return;

          // 5) Update DOM fields

          // ISBN
          const isbnValue = handleEmptyValue(variantMeta.barcode);
          isbnElement.innerHTML = `<strong>ISBN13: </strong>${isbnValue}`;

          // Weight
          const weightValue = handleEmptyValue(variantMeta.weight);
          weightElement.innerHTML = `<strong>Item Weight: </strong>${
            weightValue === 'N/A' ? 'N/A' : weightValue + ' grams'
          }`;

          // Original Price (2 decimals if valid)
          const originalPrice = handleEmptyValue(variantMeta.original_price);
          const originalCurrency = handleEmptyValue(variantMeta.original_currency);

          let displayPrice;
          if (originalPrice === 'N/A' || originalCurrency === 'N/A') {
            // If either is N/A, show N/A
            displayPrice = 'N/A';
          } else {
            // Convert to float and format to 2 decimals
            const numericPrice = parseFloat(originalPrice);
            if (isNaN(numericPrice)) {
              displayPrice = 'N/A';
            } else {
              displayPrice = `${originalCurrency} ${numericPrice.toFixed(2)}`;
            }
          }
          priceElement.innerHTML = `<strong>Original Price: </strong>${displayPrice}`;

          // Edition
          const editionValue = handleEmptyValue(variantMeta.edition);
          editionElement.innerHTML = `<strong>Edition: </strong>${editionValue}`;

          // Binding
          const bindingValue = handleEmptyValue(variantMeta.binding);
          bindingElement.innerHTML = `<strong>Binding: </strong>${bindingValue}`;
        })
        .catch((error) => {
          console.error('Error fetching variant data:', error);
        });
    }

    // --- Attach 'change' event listeners to each variant radio ---
    if (variantRadioButtons.length > 0) {
      variantRadioButtons.forEach(function (radio) {
        radio.addEventListener('change', function () {
          updateVariantData(radio.value); // same logic as on page load
        });
      });

      // --- Trigger the default variant logic on page load ---
      // 1) Find the currently checked radio, if any
      const defaultRadio = Array.from(variantRadioButtons).find((r) => r.checked);

      // 2) If found, update using its value; otherwise, fallback to the first radio button
      if (defaultRadio) {
        updateVariantData(defaultRadio.value);
      } else {
        // If no variant is preselected, pick the first
        updateVariantData(variantRadioButtons[0].value);
        // Optionally set it as checked in the DOM:
        variantRadioButtons[0].checked = true;
      }
    }
  });
</script>
